pipeline {

    agent any

    environment {
        SONARQUBE = 'sonarqube'
        SONAR_PROJECT_KEY = 'sonarqube-repository'

        IMAGE_NAME = 'my-tomcat:latest'
        CONTAINER_NAME = 'tomcat'
    }

    stages {

        stage('Checkout') {
            steps {
                 git branch: '**',
                    url: 'https://github.com/Daniyal1Hazari/jenkins-sonarqube-ci-cd.git'
    }
}

        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }

        stage('SonarQube Analysis') {
            steps {
                withSonarQubeEnv("${SONARQUBE}") {
                    sh """
                        mvn sonar:sonar \
                        -Dsonar.projectKey=${SONAR_PROJECT_KEY}
                    """
                }
            }
        }

        stage('Quality Gate') {
            steps {
                timeout(time: 10, unit: 'MINUTES') {
                    waitForQualityGate abortPipeline: true
                }
            }
        }

        stage('Docker Build & Run') {
            steps {
                sh '''
                    docker build -t ${IMAGE_NAME} .

                    docker rm -f ${CONTAINER_NAME} 2>/dev/null || true

                    docker run -d \
                        --name ${CONTAINER_NAME} \
                        -p 8081:8080 \
                        ${IMAGE_NAME}
                '''
            }
        }

        stage('Validation Successful') {
            steps {
                echo '========================================'
                echo 'VALIDATION SUCCESSFUL'
                echo '========================================'
                echo 'Maven Build       : PASSED'
                echo 'SonarQube         : PASSED'
                echo 'Quality Gate      : PASSED'
                echo 'Docker Build      : PASSED'
                echo 'Tomcat Container  : RUNNING'
                echo 'Application Port  : 8080'
                echo '========================================'
            }
        }
    }

    post {
        failure {
            echo 'VALIDATION FAILED - Pipeline stopped.'
        }

        always {
            cleanWs()
        }
    }
}
